\addcontentsline{toc}{subsection}{Figure S2 --- Funnel plots, Hedges' \textit{g}}
\begin{figure}[H]
\begin{center}
\includegraphics[width=6in]{figures/funnel_hedges.pdf}
\end{center}
\caption{\textbf{Funnel plots of Hedges' g data.} Both panels plot each effect size against its precision ($1/SE$), but differ in what is removed before plotting: (\textbf{a}) Marginal residuals are the deviations of each observed effect from the fixed-effect (intercept, the overall mean) prediction, so they still carry all of the random-effects heterogeneity (between-study, between-effect-size, and between-species--\textit{i.e.} the hierarchical non-independence of effect sizes). Asymmetry here (defined in Methods) is the classic visual signature of small-study and publication bias. (\textbf{b}) Conditional residuals additionally subtract the estimated random intercepts (study, effect-size, and species BLUPs), so the heterogeneity captured by the multilevel structure is partialled out. A symmetric, evenly-filled funnel here means that once the multilevel structure is accounted for, there is little precision-related asymmetry left, \textit{i.e.}, the asymmetry seen in (\textbf{a}) is largely explained by the modelled random effects rather than by selective reporting. Any asymmetry that persists in (\textbf{b}) is the more credible visual indication of true small-study effects.}
\label{fig:supplementaryfunnelhedges}
\end{figure}

\addcontentsline{toc}{subsection}{Figure S4 --- Leave-one-study-out, Hedges' \textit{g}}
\begin{figure}[H]
\centering
\includegraphics[width=\linewidth,height=0.7\textheight,keepaspectratio]{figures/loo_study_hedges.pdf}
\begingroup
\small
\caption[Leave-one-study-out analysis (Hedges' \textit{g})]{\textbf{Leave-one-study-out analysis of the combined Hedges' \textit{g}.} Each row re-estimates the multilevel pooled mean after removing one study from the continuous (Hedges' \textit{g}) dataset, with rows ordered by the leave-one-out estimate so the panel reads as an influence ranking. Grey-filled points are the effect sizes retained in that fit, and open orange rings are the effect sizes that were removed, each scaled by its precision ($1/SE$). The white marker is the re-estimated pooled mean, the thick black bar its 95\% confidence interval and the thin black whisker its 95\% prediction interval. The two vertical dashed red lines mark the full-sample 95\% confidence interval, so a mean falling between them indicates that omitting that study leaves the pooled estimate essentially unchanged. Labels on the right give the number of effect sizes retained ($k$), with the number of studies remaining in parentheses.}
\endgroup
\label{fig:supplementaryloostudyhedges}
\end{figure}

\caption{\textbf{Sensitivity of the combined Hedges' \textit{g} to the contentious effect sizes.} Each row shows the combined random-effects estimate for the continuous (Hedges' \textit{g}) dataset under one version of the data: the original analysis (\textit{Main}), the conservative version with all seven flagged studies removed (\textit{Leave-out}), and the best-estimate version with their effect sizes replaced by re-extractions (\textit{Corrected}).
